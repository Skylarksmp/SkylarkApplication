/* =====================================================
   EMAIL SECTION HELPER
===================================================== */

function createSection(title, content) {

    return (

        "<div style=\"" +
        "margin-top:22px;" +
        "padding:20px;" +
        "background:#06263d;" +
        "border:1px solid #164d66;" +
        "border-radius:12px;" +
        "\">" +

            "<h3 style=\"" +
            "color:#67eaff;" +
            "font-size:14px;" +
            "margin-top:0;" +
            "\">" +

            escapeHtml(title) +

            "</h3>" +

            "<p style=\"" +
            "color:#c2dce6;" +
            "font-size:14px;" +
            "line-height:1.7;" +
            "white-space:pre-wrap;" +
            "\">" +

            escapeHtml(content) +

            "</p>" +

        "</div>"

    );

}


/* =====================================================
   HTML ESCAPE
===================================================== */

function escapeHtml(value) {

    return String(value)

        .replace(/&/g, "&amp;")

        .replace(/</g, "&lt;")

        .replace(/>/g, "&gt;")

        .replace(/"/g, "&quot;")

        .replace(/'/g, "&#039;");

}
